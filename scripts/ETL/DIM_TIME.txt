SET LANGUAGE SPANISH;
DECLARE @FECHA_COMPLETA AS DATETIME, 
@ANIO AS CHAR(4), @MES AS SMALLINT, @MES_LETRA AS CHAR(10),
@SEMESTRE AS SMALLINT, @TRIMESTRE AS SMALLINT, @SEMESTRE_LETRA AS CHAR(20), @TRIMESTRE_LETRA AS CHAR(20),
@DIA AS SMALLINT, 
@SEMANA AS SMALLINT, @SEMANA_LETRA AS CHAR(20),
@QUINCENA AS SMALLINT, @QUINCENA_LETRA AS CHAR(20),
@DIA_LETRA AS CHAR(10),@HORA AS CHAR(2), @MINUTO AS CHAR(2), @SEGUNDO AS CHAR(2),
@FECHA_INICIO AS DATETIME, @FECHA_TERMINO AS DATETIME
BEGIN TRANSACTION
SELECT @FECHA_INICIO = CAST('01/01/2022 00:00:00' AS DATETIME)
SELECT @FECHA_TERMINO = CAST('31/12/2025 00:00:00' AS DATETIME)
WHILE (@FECHA_INICIO <= @FECHA_TERMINO) 
BEGIN
	SELECT @FECHA_COMPLETA = @FECHA_INICIO
	SELECT @ANIO = DATEPART(YYYY, @FECHA_INICIO)
	SELECT @TRIMESTRE = DATEPART(QUARTER, @FECHA_INICIO)
	SET @TRIMESTRE_LETRA = (SELECT CASE @TRIMESTRE WHEN 1 THEN 'Primer Trimestre' WHEN 2 THEN 'Segundo Trimestre' WHEN 3 THEN 'Tercer Trimestre' WHEN 4 THEN 'Cuarto Trimestre' END)
	SELECT @SEMESTRE = ((DATEPART(MM, @FECHA_INICIO) -1 ) / 6 ) + 1
	SET @SEMESTRE_LETRA = (SELECT CASE @SEMESTRE WHEN 1 THEN 'Primer Semestre' WHEN 2 THEN 'Segundo Semestre' END)
	SELECT @MES = DATEPART(MM, @FECHA_INICIO)
	SELECT @MES_LETRA = DATENAME(MM, @FECHA_INICIO)
	SELECT @QUINCENA = (((DATEPART(DD, @FECHA_INICIO)) - 1)/15) + 1
	IF @QUINCENA = 3 
	BEGIN
		SET @QUINCENA = @QUINCENA - 1
	END
	SET @QUINCENA_LETRA = (SELECT CASE @QUINCENA WHEN 1 THEN 'Primera Quincena' WHEN 2 THEN 'Segunda Quincena' END)
	
	SELECT @DIA = DATEPART(DD, @FECHA_INICIO)
	SELECT @DIA_LETRA = DATENAME(DW,@FECHA_INICIO)

	SELECT @SEMANA = datepart(week,@FECHA_INICIO) - datepart(week, dateadd(dd,-day(@FECHA_INICIO)+1,@FECHA_INICIO))+1



	SET @SEMANA_LETRA = (
		SELECT CASE @SEMANA 
			WHEN 1 THEN 'Primera Semana' 
			WHEN 2 THEN 'Segunda Semana' 
			WHEN 3 THEN 'Tercera Semana'
			WHEN 4 THEN 'Cuarta Semana'
			WHEN 5 THEN 'Quinta Semana'
			WHEN 6 THEN 'Sexta Semana'
			ELSE 'N/A'
		END
	)
	INSERT INTO TurismoRealDM.DBO.DIM_TIME
	(date_desc, year_desc, trimester_id, trimester_desc, semester_id, semester_desc, month_id, month_desc, fortnight_id, fortnight_desc, week_id, week_desc, day_id, day_desc)
	VALUES 
	(@fecha_completa, @anio, @TRIMESTRE, @TRIMESTRE_LETRA, @SEMESTRE, @SEMESTRE_LETRA, @MES, @MES_LETRA, @QUINCENA, @QUINCENA_LETRA, @SEMANA, @SEMANA_LETRA, @DIA,@DIA_LETRA);
	SELECT @FECHA_INICIO =  DATEADD(DAY,1,@FECHA_INICIO)
END
COMMIT TRANSACTION
