SELECT DIM_CLIENT.id AS DIM_CLIENT_ID, DIM_DWELLING.id AS DIM_DWELLING_ID, DIM_TIME.id AS TIME_ID, DIM_ORDER.id AS DIM_ORDER, ORDER_REL.FECHA_INICIO, ORDER_REL.FECHA_TERMINO, ORDER_REL.ABONO, 
                  ORDER_REL.TOTAL_PAGO, ORDER_REL.MONTO_PAGADO, ORDER_REL.CANT_TOTAL
FROM     TURISMO_REAL.RESERVA AS ORDER_REL INNER JOIN
                  TURISMO_REAL.CLIENTE AS CLIENT_REL ON CLIENT_REL.ID = ORDER_REL.ID_CLI INNER JOIN
                  TURISMO_REAL.PERSONA AS PERSON_REL ON CLIENT_REL.ID = PERSON_REL.ID INNER JOIN
                  CountriesOLTP.paises.countries AS COUNTRY_REL ON PERSON_REL.ID_PAI = COUNTRY_REL.id INNER JOIN
                  CountriesOLTP.paises.states AS COUNTRY_STATE_REL ON PERSON_REL.ID_EST = COUNTRY_STATE_REL.id INNER JOIN
                  CountriesOLTP.paises.cities AS CITY_REL ON PERSON_REL.ID_CIU = CITY_REL.id INNER JOIN
                  TURISMO_REAL.ESTADO_CIVIL AS MARITAL_STATUS_REL ON PERSON_REL.ID_EST1 = MARITAL_STATUS_REL.ID INNER JOIN
                  TURISMO_REAL.GENERO AS GENRE_REL ON PERSON_REL.ID_GEN = GENRE_REL.ID INNER JOIN
                  TURISMO_REAL.DOC_IDENTIDAD AS IDENTITY_REL ON PERSON_REL.ID_DOC = IDENTITY_REL.ID INNER JOIN
                  TURISMO_REAL.VIVIENDA AS DWELLING_REL ON ORDER_REL.ID_VIV = DWELLING_REL.ID INNER JOIN
                  TURISMO_REAL.TIPO_VIVIENDA AS CATEGORY_REL ON DWELLING_REL.ID_TIP = CATEGORY_REL.ID INNER JOIN
                  TURISMO_REAL.DISPONIBILIDAD AS AVAILABILITY_REL ON DWELLING_REL.ID_DIS = AVAILABILITY_REL.ID INNER JOIN
                  TurismoRealDM.dbo.DIM_CLIENT AS DIM_CLIENT ON DIM_CLIENT.client_id = CLIENT_REL.ID INNER JOIN
                  TurismoRealDM.dbo.DIM_DWELLING AS DIM_DWELLING ON DIM_DWELLING.dwelling_id = DWELLING_REL.ID INNER JOIN
                  TurismoRealDM.dbo.DIM_ORDER AS DIM_ORDER ON DIM_ORDER.order_id = ORDER_REL.ID INNER JOIN
                  TurismoRealDM.dbo.DIM_TIME AS DIM_TIME ON
                      (SELECT RIGHT('00' + CAST(DATEPART([DAY], CAST(DIM_TIME.date_desc AS DATE)) AS NVARCHAR(10)), 2) + '-' + RIGHT('00' + CAST(DATEPART([MONTH], CAST(DIM_TIME.date_desc AS DATE)) AS NVARCHAR(10)), 2) 
                                         + '-' + CAST(DATEPART([YEAR], CAST(DIM_TIME.date_desc AS DATE)) AS NVARCHAR(10)) AS Expr1) =
                      (SELECT RIGHT('00' + CAST(DATEPART([DAY], CAST(ORDER_REL.CREACION AS DATE)) AS NVARCHAR(10)), 2) + '-' + RIGHT('00' + CAST(DATEPART([MONTH], CAST(ORDER_REL.CREACION AS DATE)) AS NVARCHAR(10)), 2) 
                                         + '-' + CAST(DATEPART([YEAR], CAST(ORDER_REL.CREACION AS DATE)) AS NVARCHAR(10)) AS Expr1)
